const express = require('express');
const fs = require('fs');
const cors = require('cors');
const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());
app.use(express.static('public'));

// ✅ Sửa lỗi Cannot GET /
app.get("/", (req, res) => {
  res.sendFile(__dirname + "/public/index.html");
});

const dbFile = './data.json';
let db = { keys: {}, bannedIPs: {} };

if (fs.existsSync(dbFile)) {
  db = JSON.parse(fs.readFileSync(dbFile, 'utf8'));
}

function saveDB() {
  fs.writeFileSync(dbFile, JSON.stringify(db, null, 2));
}

app.post('/api/verify', (req, res) => {
  const { key, ip } = req.body;
  if (!key || !ip) return res.status(400).json({ error: 'Thiếu key hoặc IP' });

  if (db.bannedIPs[ip]?.includes(key)) {
    return res.status(403).json({ error: 'IP này đã bị cấm do sai quá 5 lần!' });
  }

  const data = db.keys[key];
  if (!data) {
    db.bannedIPs[ip] = db.bannedIPs[ip] || [];
    db.bannedIPs[ip].push(key);
    const count = db.bannedIPs[ip].filter(k => k === key).length;
    saveDB();
    if (count >= 5) return res.status(403).json({ error: 'Sai quá 5 lần. IP bị cấm.' });
    return res.status(401).json({ error: `Key không tồn tại. Sai ${count}/5 lần.` });
  }

  if (!data.ip) {
    data.ip = ip;
    saveDB();
    return res.json({ success: true, msg: 'Đăng nhập thành công lần đầu!' });
  }

  if (data.ip === ip) {
    return res.json({ success: true, msg: 'Đăng nhập đúng thiết bị.' });
  }

  return res.status(403).json({ error: 'Key đã dùng trên thiết bị khác!' });
});

app.post('/api/create', (req, res) => {
  const { key } = req.body;
  if (!key) return res.status(400).json({ error: 'Thiếu key' });
  db.keys[key] = { ip: null };
  saveDB();
  res.json({ success: true });
});

app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
